delta_squire = {
	inline_script = {
		script = pop_jobs/xhk_acot_knights/squire
		NUM_AMENITIES = 400
		CURSE_GROWTH = 0.125
		UPKEEP_BASE = "
			alloys = 1
		"
		UPKEEP_CURSE = "
			alloys = 1
		"
		PLANET_MOD = "
			planet_defense_armies_add = 1
			pop_knight_bonus_workforce_mult = 0.0275
			planet_knights_upkeep_mult = 0.005
			pop_knight_commander_bonus_workforce_mult = 0.015
		"
		
	}
	swappable_data = {
		default = {
			condition_string = WORKER_JOB_TRIGGER
			building_icon = building_order_keep
		}
	}

	possible = {
		hidden_trigger = { exists = owner }
		owner = {
			is_gestalt = no
		}
		exists = planet
		planet = {
			OR = {
				has_active_building = building_order_keep
				has_active_building = building_order_castle
			}
			OR = {
				has_building = building_acot_dm_ancient_hive_cluster
				has_building = building_acot_dm_ancient_control_center
				has_building = building_acot_dm_ancient_palace
			}
		}
	}
}
