delta_squire = {
	category = worker
	condition_string = WORKER_JOB_TRIGGER
	building_icon = building_order_keep
	clothes_texture_index = 4

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
		is_enslaved = no
	}

	possible_precalc = can_fill_specialist_job #deliberate, they are baby knights

	possible = {
		hidden_trigger = { exists = owner }
		owner = {
			is_gestalt = no
		}
	}

	planet_modifier = {
		knights_independent_category_produces_mult = 0.0225
	}

	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		planet_amenities_add = 4
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}

	triggered_planet_modifier = {
		potential = {
			planet = { has_modifier = syamelles_curse_habitat }
		}
		planet_pop_assembly_organic_add = 0.25
	}

	resources = {
		category = planet_squires
		upkeep = {
			trigger = {
				planet = { has_modifier = syamelles_curse_habitat }
			}
			alloys = 1
		}
	}

	weight = {
		weight = @worker_job_weight
		factor = 1.2
	}
}

delta_knight = {
	category = specialist
	condition_string = SPECIALIST_JOB_TRIGGER
	building_icon = building_order_keep
	clothes_texture_index = 3

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_enslaved = no
	}

	possible_precalc = can_fill_specialist_job
	possible = {
		hidden_trigger = { exists = owner }
		owner = {
			is_gestalt = no
		}
	}

	resources = {
		category = knights_independent_category
		produces = {
			trigger = { exists = owner }
			unity = 12
			mult = value:knights_unity_output_modifier
		}
		produces = {
			trigger = { exists = owner }
			physics_research = 15
			engineering_research = 15
			society_research = 15
			mult = value:knights_research_output_modifier
		}
		upkeep = {
			sr_dark_matter = 1
			acot_sr_dark_energy = 1
		}
	}

	triggered_pop_modifier = {
		potential = {
			always = yes
		}
		pop_defense_armies_add = 4
		mult = value:scripted_modifier_mult|MODIFIER|knights_independent_category_produces_mult|
	}

	triggered_planet_modifier = {
		potential = {
			owner = { has_country_flag = toxoids_7090_a }
		}
		planet_amenities_add = 10
		mult = value:scripted_modifier_mult|MODIFIER|knights_independent_category_produces_mult|
	}

	triggered_country_modifier = {
		country_naval_cap_add = 10
		mult = value:scripted_modifier_mult|MODIFIER|knights_independent_category_produces_mult|
	}

	triggered_country_modifier = {
		potential = {
			owner = { has_modifier = alloy_producing_knights }
		}
		country_alloys_produces_mult = 0.02
	}

	weight = {
		weight = @high_prio_specialist_job_weight
		factor = 1.2
		add = 5 #Somehow pops preferred to work as squires than as knights. No more.
		modifier = {
			mult = value:job_weights_modifier|JOB|bureaucrat|RESOURCE|unity|
		}
		modifier = {
			mult = value:job_weights_research_modifier|JOB|researcher|
		}
	}
}
